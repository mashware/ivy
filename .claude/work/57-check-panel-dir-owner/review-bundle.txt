+#[cfg(unix)]
+fn private_to_uid(dir: &Path, uid: u32) -> io::Result<()> {
+    use std::os::unix::{ffi::OsStrExt, fs::MetadataExt};
+    // lstat("alias/.") follows alias. Strip only terminal slash/dot components;
+    // normalising interior components (especially ..) would change symlink traversal.
+    let mut bytes = dir.as_os_str().as_bytes();
+    loop {
+        if (bytes.len() > 1 && bytes.ends_with(b"/")) || bytes.ends_with(b"/.") {
+            bytes = &bytes[..bytes.len() - 1];
+        } else {
+            break;
+        }
+    }
+    let lookup = Path::new(std::ffi::OsStr::from_bytes(bytes));
+    let metadata = std::fs::symlink_metadata(lookup).map_err(|failure| {
+        let detail = if failure.kind() == io::ErrorKind::NotFound {
+            "directory is gone".to_owned()
+        } else {
+            failure.to_string()
+        };
+        io::Error::new(
+            failure.kind(),
+            format!(
+                "this console's panel could not be reached at {} ({detail})",
+                dir.display()
+            ),
+        )
+    })?;
+    if !metadata.is_dir() || metadata.uid() != uid || metadata.mode() & 0o077 != 0 {
+        return Err(io::Error::new(
+            io::ErrorKind::PermissionDenied,
+            format!("{} is not a private directory of this user", dir.display()),
+        ));
+    }
+    Ok(())
+}

fn publish(mut input: impl Read) {
    use panel_control::client::whole_identity;
    let [id, dir] = panel_contract::ENV_NAMES.map(std::env::var_os);
    if !panel_control::endpoint::BINDS || whole_identity(id.as_deref(), dir.as_deref()).is_none() {
        return;
    }
    let bytes = match read_input(&mut input, panel_contract::MAX_BYTES as usize) {
        Ok(bytes) => bytes,
        Err(message) => {
            eprintln!("ivy-session: {message}");
            std::process::exit(2);
        }
    };
    let document = match session::publish_document(&bytes) {
        Ok(document) => document,
        Err(message) => {
            eprintln!("ivy-session: {message}");
            std::process::exit(2);
        }
    };
    let line = format!("{{\"tool\":\"panel_publish\",\"document\":{document}}}\n");
    let reply = match session::exchange_line(line.as_bytes()) {
        Ok(reply) => reply,
        Err(failure) => {
            eprintln!("ivy-session: {failure}");
            std::process::exit(1);
        }
    };
    if reply["ok"] == true {
        println!("{{\"ok\":true}}");
        return;
    }
    if reply["code"] == "unheld" {
        println!("{{\"ok\":false,\"code\":\"unheld\",\"message\":\"this panel has no holder\"}}");
        return;
    }
    let message = reply["message"]
        .as_str()
        .or_else(|| reply["error"].as_str())
        .unwrap_or("publication refused");
    eprintln!("ivy-session: {message}");
    std::process::exit(1);
}

fn request(args: Vec<std::ffi::OsString>, mut input: impl Read) -> Result<Request, String> {
    let args: Vec<&str> = args
        .iter()
        .map(|arg| arg.to_str().ok_or("invalid helper arguments"))
        .collect::<Result<_, _>>()?;
    let request = match args.as_slice() {
        ["hook", "codex"] => {
            let bytes = read_input(&mut input, session::MAX_HOOK_INPUT)?;
            session::hook_request(&bytes)?
        }
        ["attach", client] => Request::Attach {
            client: Client::parse(client)?,
        },
        ["probe", generation] => Request::Probe {
            generation: (*generation).into(),
        },
        ["update", generation] => {
            let bytes = read_input(&mut input, session::MAX_REQUEST)?;
            session::update_request((*generation).into(), &bytes)?
        }
        _ => return Err(
            "usage: ivy-session publish | hook codex | attach CLIENT | probe GENERATION | update GENERATION"
                .into(),
        ),
    };
    request.validate()?;
    Ok(request)
}

fn read_input(input: &mut impl Read, limit: usize) -> Result<Vec<u8>, String> {
    let mut bytes = Vec::new();
    input
        .take(limit as u64 + 1)
        .read_to_end(&mut bytes)
        .map_err(|_| "cannot read session metadata")?;
    if bytes.len() > limit {
        return Err("session metadata goes over the ceiling".into());
    }
    Ok(bytes)
}

//! The producer's side of the channel.
//!
//! **It is here and not in whoever speaks MCP**, because what a request looks like on the wire is
//! the contract's business and not the vocabulary's: the day the same channel is spoken over
//! something else, that vocabulary is the piece that goes and this is the piece that stays.

use std::io::{self, Read, Write};
use std::path::Path;

use interprocess::local_socket::Stream;

use crate::endpoint;
use crate::wire::{MAX_REQUEST, Side};

/// The pane identity a program inherited, **when it is whole**: `PANEL_ID` present, non-empty and
/// text, and `PANEL_DIR` present and an absolute path.
///
/// **The one rule the MCP bridge applies, to its handshake and to its tool calls alike**, because
/// two rules in one process gave two answers: the handshake said «no console» for an empty
/// identifier while the first tool call connected with it. Bringing the session helper under it was
/// left out of that decision (D88). Empty counts as absent — a client configuration that
/// interpolates `${PANEL_ID:-}` produces one, and on Windows an empty identifier names a single
/// machine-wide endpoint. An identifier that is not text cannot be one this terminal generated: the
/// identity is the pair (D7), and on Windows, where the identifier names the endpoint, converting
/// it lossily would connect to a mangled name. A relative directory cannot be one either:
/// it would resolve against the working directory and reach whatever socket sits there.
///
/// **It takes the values instead of reading them** so it can be tested without `set_var`, which
/// is undefined behaviour in a suite that launches processes in parallel.
pub fn whole_identity<'a>(
    id: Option<&'a std::ffi::OsStr>,
    dir: Option<&'a std::ffi::OsStr>,
) -> Option<(&'a str, &'a Path)> {
    let id = id?.to_str().filter(|id| !id.is_empty())?;
    let dir = Path::new(dir?);
    dir.is_absolute().then_some((id, dir))
}

/// A connection to one pane's channel.
///
/// **The hold this delivery gives a source lasts as long as this value does**, which is why it is a
/// type rather than a function per request: publishing twice through two connections would claim the
/// pane twice and be refused the second time.
pub struct Producer {
    stream: Stream,
}

impl Producer {
    /// Opens the channel of the pane named by the two variables the program inherited.
    ///
    /// **No third variable, and no identifier on the wire.** Which pane this reaches is decided by
    /// which endpoint was opened, so there is nothing in any message for anybody to trust.
    ///
    /// **A whole identity or nothing, and never a fallback** ([`whole_identity`]). Which of the two
    /// variables names the endpoint is the platform's business ([`endpoint::name_of`]), so half an
    /// identity is a missing one on one platform and harmless on the other — and defaulting the
    /// identifier to the empty string resolved every pane on Windows to a single machine-wide name.
    ///
    /// **Two identity refusals, because they have two causes** — a platform with no channel and a
    /// console that cannot be reached are refused apart, after them. Neither variable carrying
    /// anything — absent or empty, which is what a client interpolating `${PANEL_ID:-}` outside any
    /// pane hands over — is a program outside any pane. One carrying something but the identity not
    /// whole is a client that forwards them badly — a list of variables missing one, a value
    /// mangled on the way — and saying «not in a pane» there would send whoever reads it looking in
    /// the wrong place.
    pub fn of_this_pane() -> io::Result<Self> {
        let [id, dir] = panel_contract::ENV_NAMES.map(std::env::var_os);
        let unset =
            |value: &Option<std::ffi::OsString>| value.as_deref().is_none_or(|v| v.is_empty());
        let outside = unset(&id) && unset(&dir);
        let (id, dir) = whole_identity(id.as_deref(), dir.as_deref()).ok_or_else(|| {
            if outside {
                io::Error::new(
                    io::ErrorKind::NotFound,
                    "this program is not running inside a pane of this terminal",
                )
            } else {
                io::Error::new(
                    io::ErrorKind::InvalidInput,
                    "PANEL_ID or PANEL_DIR is set, but the two are not a whole pane identity (one \
                     is missing or empty, the identifier is not text, or the directory is not \
                     absolute): check how your client forwards them",
                )
            }
        })?;
        // Where no channel binds, a live console is unreachable by construction: say that, not
        // that the console went away.
        if !endpoint::BINDS {
            return Err(io::Error::new(
                io::ErrorKind::Unsupported,
                "this platform has no control channel yet: publish this console's panel as a file \
                 in PANEL_DIR",
            ));
        }

        // Any failure to reach a whole identity — most often the console closed and its directory
        // gone — is said naming the directory, which the bare system error does not. Its kind is
        // chosen to differ from «outside any pane»; the original error survives only as text.
        endpoint::private_to_this_user(dir)?;
        Self::of(dir, id).map_err(|failure| {
            io::Error::new(
                io::ErrorKind::NotConnected,
                format!(
                    "this console's panel could not be reached at {} ({failure})",
                    dir.display()
                ),
            )
        })
    }

    /// The same, for a pane named directly. **What the tests use**, and what a producer that manages
    /// several panes of its own would use. It takes the pair as given: the rule of
    /// [`whole_identity`] is the caller's to apply.
    pub fn of(dir: &Path, id: &str) -> io::Result<Self> {
        Ok(Self {
            stream: endpoint::connect(dir, id)?,
        })
    }

    /// Publishes a document, and returns what the window answered.
    ///
    /// **The document is written back out on one line, and that is this side's job rather than the
    /// producer's.** The wire is framed by newlines, and the natural thing for somebody writing JSON
    /// by hand — or for any pretty-printer — is to produce several lines; interpolating that raw
    /// split the request across frames, so the pane read a first line that was not valid JSON and
    /// then a tail of nothing. Compacting here costs one parse on the producer's own thread and
    /// makes the framing impossible to break from outside.
    ///
    /// It also means a document that is not JSON at all is refused **here, to its author**, instead
    /// of travelling to the window to come back as an answer nobody can act on.
    pub fn publish(&mut self, source: &str, document: &str) -> io::Result<Answered> {
        let document: serde_json::Value = serde_json::from_str(document).map_err(|failure| {
            io::Error::new(
                io::ErrorKind::InvalidInput,
                format!("the document is not JSON ({failure})"),
            )
        })?;
        // `Value`'s `Display` is the compact form: no newline can survive it, which is the property
        // the framing rests on.
        let escaped_source = serde_json::Value::String(source.to_owned());
        writeln!(
            self.stream,
            "{{\"tool\":\"panel_set\",\"source\":{escaped_source},\"document\":{document}}}"
        )?;
        self.stream.flush()?;

        self.read_reply()
    }

    /// Opens a console beside this pane's own, and returns the handle the window gave it.
    ///
    /// **The handle is what the rest of the verbs are named by**, and it is opaque on purpose: it
    /// belongs to this connection, and a second connection cannot use it even if it reads it.
    pub fn open_a_pane(
        &mut self,
        side: Side,
        program: Option<&str>,
        args: &[String],
    ) -> io::Result<Answered> {
        // **Built as a value and not interpolated**, so a program name with a quote in it cannot
        // break the request it travels in — the same hazard the reply's escaping answers.
        let mut request = serde_json::json!({"tool": "split_open", "side": side.name()});
        if let Some(program) = program {
            request["program"] = serde_json::Value::String(program.to_owned());
        }
        if !args.is_empty() {
            request["args"] = serde_json::json!(args);
        }
        self.ask(request)
    }

    /// Reads a console this connection opened: **the visible screen unless the scrollback is asked
    /// for**, which is the contract's default.
    pub fn read_a_pane(&mut self, pane: &str, scrollback: bool) -> io::Result<Answered> {
        self.ask(serde_json::json!({
            "tool": "split_read",
            "pane": pane,
            "scrollback": scrollback,
        }))
    }

    /// Asks for the document this pane's panel is showing.
    ///
    /// **It takes no hold and needs none**: it is for the producer that has *not* claimed the pane —
    /// one that restarted, one whose context was compacted away, a second tool taking the console
    /// over — and it answers with a document that arrived by file just as readily as with one this
    /// connection published.
    pub fn read_the_document(&mut self) -> io::Result<Answered> {
        self.ask(serde_json::json!({"tool": "panel_get"}))
    }

    /// Publishes a project plan document. The endpoint's pane determines the project; `repo` is
    /// carried for producer compatibility but is intentionally ignored by the receiver.
    pub fn publish_plan(&mut self, document: &str, repo: Option<&str>) -> io::Result<Answered> {
        let document: serde_json::Value = serde_json::from_str(document).map_err(|failure| {
            io::Error::new(
                io::ErrorKind::InvalidInput,
                format!("the plan is not JSON ({failure})"),
            )
        })?;
        let mut request = serde_json::json!({"tool": "plan_set", "document": document});
        if let Some(repo) = repo {
            request["repo"] = serde_json::Value::String(repo.to_owned());
        }
        self.ask(request)
    }

    /// Reads the published project plan for this pane's project.
    pub fn read_plan(&mut self, repo: Option<&str>) -> io::Result<Answered> {
        let mut request = serde_json::json!({"tool": "plan_get"});
        if let Some(repo) = repo {
            request["repo"] = serde_json::Value::String(repo.to_owned());
        }
        self.ask(request)
    }

    /// Retires the published project plan for this pane's project.
    pub fn retire_plan(&mut self, repo: Option<&str>) -> io::Result<Answered> {
        let mut request = serde_json::json!({"tool": "plan_retire"});
        if let Some(repo) = repo {
            request["repo"] = serde_json::Value::String(repo.to_owned());
        }
        self.ask(request)
    }

    /// Changes part of the document this connection published, leaving the rest of it alone.
    ///
    /// **It needs the hold that a publication takes**, unlike [`Producer::read_the_document`]: there
    /// is no state where a connection holds a pane and there is nothing to patch, so *nothing has
    /// been published yet* and *this pane's panel is its file's* are one refusal, and it names the
    /// move — publish first.
    ///
    /// **What happens here is the framing, not the reading.** The ops are checked to be JSON and
    /// compacted, for the same two reasons a document is: a list that is not JSON at all is refused
    /// to its author instead of travelling to the window, and a compact line cannot carry the newline
    /// the framing rests on. **What each op says is decided at the other end** — by
    /// `panel_contract::parse_ops_noting`, on the connection's thread inside the application — so an op
    /// nobody serves or a field that is not patchable travels and comes back as a refusal.
    pub fn patch(&mut self, ops: &str) -> io::Result<Answered> {
        let ops: serde_json::Value = serde_json::from_str(ops).map_err(|failure| {
            io::Error::new(
                io::ErrorKind::InvalidInput,
                format!("the ops are not JSON ({failure})"),
            )
        })?;
        self.ask(serde_json::json!({"tool": "panel_patch", "ops": ops}))
    }

    /// Intentionally ends the session currently holding this pane's panel.
    pub fn retire(&mut self) -> io::Result<Answered> {
        self.ask(serde_json::json!({"tool": "panel_retire"}))
    }

    /// Closes a console this connection opened.
    pub fn close_a_pane(&mut self, pane: &str) -> io::Result<Answered> {
        self.ask(serde_json::json!({"tool": "split_close", "pane": pane}))
    }

    /// Writes one request line and reads its answer, **the one place that talks to the socket**.
    ///
    /// `Value`'s `Display` is the compact form, so a request can never carry a newline and break the
    /// framing the wire rests on.
    fn ask(&mut self, request: serde_json::Value) -> io::Result<Answered> {
        writeln!(self.stream, "{request}")?;
        self.stream.flush()?;

        self.read_reply()
    }

    /// Reads one reply line, bounded the same way a request is.
    fn read_reply(&mut self) -> io::Result<Answered> {
        let mut line = Vec::new();
        let mut byte = [0_u8; 1];
        loop {
            match self.stream.read(&mut byte)? {
                0 => {
                    return Err(io::Error::new(
                        io::ErrorKind::UnexpectedEof,
                        "the pane closed the channel without answering",
                    ));
                }
                _ if byte[0] == b'\n' => break,
                _ => line.push(byte[0]),
            }
            if line.len() > MAX_REQUEST {
                return Err(io::Error::other("the answer goes over the ceiling"));
            }
        }

        // **The document is kept as the bytes that arrived, and every other field is not.** A
        // `serde_json::Value` cannot do it: without `preserve_order` its map is sorted, so a document
        // published as `{"updated_at":…,"phase":…,"lines":…}` came back
        // `{"lines":…,"phase":…,"updated_at":…}` — the same panel, and not the same bytes. That
        // defeats the one reason the reply carries the document unescaped, and it made this type's own
        // promise —«as the JSON text it arrived as»— false. Borrowing it as a [`RawValue`] is the same
        // answer the request side already gives for the same reason.
        let raw: Reply<'_> = serde_json::from_slice(&line)
            .map_err(|failure| io::Error::other(failure.to_string()))?;
        // **The field travels, not its name.** Looking one up by a string would put a `_ => None` arm
        // between a refusal and the caller: a typo would compile, every refusal would collapse into
        // «it refused without saying why», and the reason would be thrown away — the same silent loss
        // the branch below exists to remove.
        let text_of = |held: Option<&serde_json::value::RawValue>| -> Option<String> {
            held.and_then(|held| serde_json::from_str::<String>(held.get()).ok())
        };
        match raw.ok {
            Some(true) if raw.closed.is_some() || raw.preserved.is_some() => {
                match (raw.closed, raw.preserved) {
                    (Some(closed), Some(preserved)) => Ok(Answered::Retired { closed, preserved }),
                    _ => Ok(Answered::Failed(
                        "the pane answered retirement without both counts".to_owned(),
                    )),
                }
            }
            Some(true) => match raw.pane {
                // **Present and possibly null**: an open answers with the handle, or with a JSON null
                // where the console has no addressable identity, and those are two different
                // answers a caller can act on differently. A handle that is neither text nor null
                // would also land on `None`; the wire writes only those two (`Reply::Opened` in
                // `wire.rs`), so no separate answer is kept for it.
                Some(pane) => Ok(Answered::Opened(text_of(Some(pane)))),
                // **Before the fallback and not after it.** Everything that is not `pane` and not
                // `text` used to land on `Answered::Ok` — success carrying nothing — so a reply whose
                // whole content is the document would have read as «it worked» with the document
                // thrown away, and silently.
                None => match raw.document {
                    Some(document) => Ok(Answered::Document(document.get().to_owned())),
                    // Before the fallback for the reason the arm above gives: landing on `Ok` is a
                    // lossy publication answered as a perfect one, which is the silence this field
                    // exists to break. A note that is not text still says something fell, so it is
                    // handed over as the JSON it arrived as rather than dropped into a plain success.
                    None => match raw.dropped {
                        Some(dropped) => Ok(Answered::Lossy(
                            text_of(Some(dropped)).unwrap_or_else(|| dropped.get().to_owned()),
                        )),
                        None => match text_of(raw.text) {
                            Some(text) => Ok(Answered::Text(text)),
                            None => Ok(Answered::Ok),
                        },
                    },
                },
            },
            _ => Ok(Answered::Failed(text_of(raw.error).unwrap_or_else(|| {
                "the pane refused it without saying why".to_owned()
            }))),
        }
    }
}

/// One reply line, **with the document kept as the bytes it arrived as**.
///
/// Every JSON-valued field is borrowed raw for the reason the request's own envelope states: this crate does not
/// re-serialise somebody else's JSON, because re-serialising is what silently reordered a document's
/// keys on the way back to its producer.
#[derive(serde::Deserialize)]
struct Reply<'a> {
    ok: Option<bool>,
    #[serde(default)]
    closed: Option<usize>,
    #[serde(default)]
    preserved: Option<usize>,
    /// **A null `pane` is an answer of its own** — the console was opened with no addressable
    /// identity — so it must not be read as an absent field, which is what `Option`'s own
    /// `Deserialize` does with a null. Hence [`present`].
    #[serde(default, borrow, deserialize_with = "present")]
    pane: Option<&'a serde_json::value::RawValue>,
    #[serde(default, borrow)]
    text: Option<&'a serde_json::value::RawValue>,
    #[serde(default, borrow)]
    document: Option<&'a serde_json::value::RawValue>,
    #[serde(default, borrow)]
    dropped: Option<&'a serde_json::value::RawValue>,
    #[serde(default, borrow)]
    error: Option<&'a serde_json::value::RawValue>,
}

/// `Some` for a field that is there, **null included** — serde only calls this for a present field,
/// and `default` supplies the `None` for an absent one.
fn present<'de, D>(field: D) -> Result<Option<&'de serde_json::value::RawValue>, D::Error>
where
    D: serde::Deserializer<'de>,
{
    serde::Deserialize::deserialize(field).map(Some)
}

/// What a pane answered.
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Answered {
    /// It was applied, and the reply carried nothing else.
    Ok,
    /// It was applied, and **part of what was sent fell away on the way in**: the sentence naming
    /// what, in the contract's words.
    Lossy(String),
    /// A session was retired, with its auxiliary-console outcome.
    Retired {
        closed: usize,
        preserved: usize,
    },
    /// A console was opened; the handle is `None` where it has no addressable identity.
    Opened(Option<String>),
    /// What a console showed.
    Text(String),
    /// The document a console's panel is showing, **as the bytes it arrived as** — not a
    /// re-serialisation of them, which sorts the keys and hands the producer a document it did not
    /// publish.
    Document(String),
    Failed(String),
}

//! Where a pane's channel lives, and who is allowed to reach it.
//!
//! `PanelDir` creates a `0o700` directory. Inherited clients also verify its owner and
//! permissions before connecting: a stale identity may name a replacement planted after the
//! console closed. This pathname check does not authenticate the peer across a replacement race.

use std::io;
use std::path::Path;

#[cfg(unix)]
use std::os::fd::AsRawFd;
#[cfg(unix)]
use std::os::unix::net::UnixStream;

#[cfg(not(windows))]
use interprocess::local_socket::GenericFilePath;
#[cfg(windows)]
use interprocess::local_socket::GenericNamespaced;
use interprocess::local_socket::{Listener, Stream, prelude::*};
use interprocess::local_socket::{ListenerOptions, Name};

/// The channel-owned end of a private wake pair.
///
/// Dropping it makes the acceptor's end readable with a hang-up. It needs neither the endpoint's
/// pathname nor room in its connection backlog, and closing a descriptor cannot wait.
pub(crate) struct AcceptorStopper {
    #[cfg(unix)]
    _signal: UnixStream,
}

/// The listener and private wake descriptor owned by one acceptor thread.
pub(crate) struct Acceptor {
    listener: Listener,
    #[cfg(unix)]
    signal: UnixStream,
}

/// What woke the acceptor.
pub(crate) enum AcceptorReady {
    Connection,
    Stop,
}

/// Gives one bound listener the private descriptor pair that stops it.
pub(crate) fn acceptor(listener: Listener) -> io::Result<(AcceptorStopper, Acceptor)> {
    #[cfg(windows)]
    {
        Err(io::Error::new(
            io::ErrorKind::Unsupported,
            "the control channel has no Windows listener until F2.7",
        ))
    }
    #[cfg(unix)]
    {
        let (stopper, wake) = UnixStream::pair()?;
        Ok((
            AcceptorStopper { _signal: stopper },
            Acceptor {
                listener,
                signal: wake,
            },
        ))
    }
}

impl Acceptor {
    /// Sleeps until the endpoint has a connection or its owner disappears.
    pub(crate) fn wait(&self) -> io::Result<AcceptorReady> {
        #[cfg(windows)]
        {
            Err(io::Error::new(
                io::ErrorKind::Unsupported,
                "the control channel has no Windows listener until F2.7",
            ))
        }
        #[cfg(unix)]
        {
            let Listener::UdSocket(listener) = &self.listener;
            let mut descriptors = [
                libc::pollfd {
                    fd: listener.inner().as_raw_fd(),
                    events: libc::POLLIN,
                    revents: 0,
                },
                libc::pollfd {
                    fd: self.signal.as_raw_fd(),
                    events: libc::POLLIN,
                    revents: 0,
                },
            ];
            // SAFETY: both descriptors remain owned for the call, the array has the declared
            // length, and `poll` only writes its `revents` fields.
            let ready = unsafe { libc::poll(descriptors.as_mut_ptr(), descriptors.len() as _, -1) };
            if ready < 0 {
                return Err(io::Error::last_os_error());
            }
            if descriptors[1].revents != 0 {
                Ok(AcceptorReady::Stop)
            } else {
                Ok(AcceptorReady::Connection)
            }
        }
    }

    /// Accepts the connection whose readiness [`Self::wait`] reported.
    pub(crate) fn accept(&self) -> io::Result<Stream> {
        self.listener.accept()
    }
}

/// The name of a pane's endpoint.
///
/// **A name and not a path**, because on Windows there is no path: a pipe lives in the machine's
/// namespace. The two platforms differ here and only here, which is what keeps everything above
/// this function written once.
///
/// **Which of the two arguments does the separating is the platform's business too.** On POSIX the
/// socket is a file inside the pane's own directory, so the directory is what keeps two panes apart
/// and the identifier is not read at all; on Windows the name carries no directory, so the
/// identifier is the whole of it. That is why the producer refuses to start with either variable
/// missing instead of defaulting one — see [`crate::Producer::of_this_pane`].
pub fn name_of(dir: &Path, id: &str) -> io::Result<Name<'static>> {
    #[cfg(not(windows))]
    {
        let _ = id;
        dir.join(panel_contract::CONTROL_SOCKET)
            .to_fs_name::<GenericFilePath>()
    }
    #[cfg(windows)]
    {
        let _ = dir;
        format!("ivy-{id}").to_ns_name::<GenericNamespaced>()
    }
}

/// Whether [`bind`] binds on this platform: a producer that has to decide whether to promise a
/// channel — the MCP bridge's handshake, and `Producer::of_this_pane`, which refuses before
/// connecting — reads it here instead of restating the platform list.
/// The day F2.7 binds on Windows this has to change with it, and `binding_is_what_binds_says`
/// fails there until it does.
pub const BINDS: bool = cfg!(not(windows));

/// Binds a pane's endpoint, ready to accept.
///
/// **It does not bind on Windows, and that is a decision rather than an omission.** A named pipe
/// gets a default security descriptor that lets any process on the machine open it, so binding one
/// without an explicit DACL would turn a channel whose whole access control is a `0o700` directory
/// into a machine-wide one — silently, and on the one platform where a non-local caller can exist
/// at all. The delivery that brings this crate cannot compile or run a line of Windows code, so the
/// descriptor cannot be asserted; the rule it was given is that what cannot be proven is not bound.
/// **F2.7 is the milestone that compiles here for the first time**, and it is where the pipe, its
/// descriptor and the assertion over both arrive together. Until then a Windows pane has no channel
/// and publishes by file, exactly as every pane did before this milestone.
///
/// **The listener blocks in `accept`, and that is the shape that costs nothing at rest.** An
/// earlier version made the accept non-blocking so the acceptor could poll its stop flag, which cost
/// every pane a thread waking five times a second for the life of the window — four idle panes
/// alone were about twenty wake-ups a second, and on a laptop that is what shows up as battery. The
/// acceptor parks in `poll` over the listener and a private wake descriptor; dropping the other end
/// wakes it without polling at rest or opening a client connection. The streams it hands out are
/// blocking, which is what lets a receive timeout be put on them (see [`crate::listener`]).
pub fn bind(dir: &Path, id: &str) -> io::Result<Listener> {
    // `BINDS` says the same thing to callers that must not promise a channel; the test
    // `binding_is_what_binds_says` ties the two together.
    #[cfg(windows)]
    {
        let _ = (dir, id);
        return Err(io::Error::new(
            io::ErrorKind::Unsupported,
            "the control channel is not bound on Windows until its security descriptor can be \
             asserted",
        ));
    }
    #[cfg(not(windows))]
    {
        ListenerOptions::new().name(name_of(dir, id)?).create_sync()
    }
}

/// Refuses an inherited directory that another user could control.
pub fn private_to_this_user(dir: &Path) -> io::Result<()> {
    #[cfg(unix)]
    {
        // SAFETY: geteuid has no arguments or memory preconditions.
        private_to_uid(dir, unsafe { libc::geteuid() })
    }
    #[cfg(not(unix))]
    {
        let _ = dir;
        Ok(())
    }
}

#[cfg(unix)]
fn private_to_uid(dir: &Path, uid: u32) -> io::Result<()> {
    use std::os::unix::{ffi::OsStrExt, fs::MetadataExt};
    // lstat("alias/.") follows alias. Strip only terminal slash/dot components;
    // normalising interior components (especially ..) would change symlink traversal.
    let mut bytes = dir.as_os_str().as_bytes();
    loop {
        if (bytes.len() > 1 && bytes.ends_with(b"/")) || bytes.ends_with(b"/.") {
            bytes = &bytes[..bytes.len() - 1];
        } else {
            break;
        }
    }
    let lookup = Path::new(std::ffi::OsStr::from_bytes(bytes));
    let metadata = std::fs::symlink_metadata(lookup).map_err(|failure| {
        let detail = if failure.kind() == io::ErrorKind::NotFound {
            "directory is gone".to_owned()
        } else {
            failure.to_string()
        };
        io::Error::new(
            failure.kind(),
            format!(
                "this console's panel could not be reached at {} ({detail})",
                dir.display()
            ),
        )
    })?;
    if !metadata.is_dir() || metadata.uid() != uid || metadata.mode() & 0o077 != 0 {
        return Err(io::Error::new(
            io::ErrorKind::PermissionDenied,
            format!("{} is not a private directory of this user", dir.display()),
        ));
    }
    Ok(())
}

/// Opens a connection to a pane's endpoint, from the producer's side.
pub fn connect(dir: &Path, id: &str) -> io::Result<Stream> {
    Stream::connect(name_of(dir, id)?)
}

/// Validates and compacts a hook panel document before any endpoint connection is attempted.
pub fn publish_document(bytes: &[u8]) -> Result<String, String> {
    if bytes.len() as u64 > panel_contract::MAX_BYTES {
        return Err(format!(
            "publication input goes over {} bytes",
            panel_contract::MAX_BYTES
        ));
    }
    let value: serde_json::Value = serde_json::from_slice(bytes)
        .map_err(|_| "publication input is not valid JSON".to_owned())?;
    if !value.is_object() {
        return Err("publication input must be a JSON object".to_owned());
    }
    let compact = serde_json::to_string(&value).map_err(|_| "cannot encode publication input")?;
    if compact.len() as u64 > panel_contract::MAX_BYTES {
        return Err(format!(
            "publication document goes over {} bytes",
            panel_contract::MAX_BYTES
        ));
    }
    Ok(compact)
}

/// Exchanges a prebuilt, bounded newline JSON request over this pane's inherited endpoint.
pub fn exchange_line(line: &[u8]) -> io::Result<serde_json::Value> {
    let outside = || {
        io::Error::new(
            io::ErrorKind::NotFound,
            "this helper is not running inside an Ivy pane",
        )
    };
    let dir = std::env::var_os(panel_contract::ENV_NAMES[1])
        .filter(|v| !v.is_empty())
        .ok_or_else(outside)?;
    let id = std::env::var_os(panel_contract::ENV_NAMES[0])
        .filter(|v| !v.is_empty())
        .ok_or_else(outside)?;
    crate::endpoint::private_to_this_user(std::path::Path::new(&dir))?;
    let deadline = Instant::now() + Duration::from_secs(2);
    let mut stream = interprocess::local_socket::ConnectOptions::new()
        .name(crate::endpoint::name_of(
            std::path::Path::new(&dir),
            &id.to_string_lossy(),
        )?)
        .wait_mode(interprocess::ConnectWaitMode::Timeout(remaining(deadline)?))
        .connect_sync()?;
    let mut unwritten = line;
    while !unwritten.is_empty() {
        stream.set_send_timeout(Some(remaining(deadline)?))?;
        match stream.write(unwritten) {
            Ok(0) => {
                return Err(io::Error::new(
                    io::ErrorKind::WriteZero,
                    "session endpoint stopped reading",
                ));
            }
            Ok(written) => unwritten = &unwritten[written..],
            Err(failure) if failure.kind() == io::ErrorKind::Interrupted => continue,
            Err(failure) => return Err(failure),
        }
    }
    let mut bytes = Vec::new();
    let mut chunk = [0; 1024];
    loop {
        stream.set_recv_timeout(Some(remaining(deadline)?))?;
        let read = match stream.read(&mut chunk) {
            Ok(0) => {
                return Err(io::Error::new(
                    io::ErrorKind::UnexpectedEof,
                    "session endpoint closed without a reply",
                ));
            }
            Ok(read) => read,
            Err(failure) if failure.kind() == io::ErrorKind::Interrupted => continue,
            Err(failure) => return Err(failure),
        };
        let end = chunk[..read].iter().position(|byte| *byte == b'\n');
        let content = end.unwrap_or(read);
        if bytes.len() + content > MAX_REPLY {
            return Err(io::Error::other("session reply goes over the ceiling"));
        }
        bytes.extend_from_slice(&chunk[..content]);
        if end.is_some() {
            break;
        }
    }
    let reply: serde_json::Value =
        serde_json::from_slice(&bytes).map_err(|_| io::Error::other("invalid session reply"))?;
    if !reply.is_object() || !reply["ok"].is_boolean() {
        return Err(io::Error::other("invalid session reply"));
    }
    Ok(reply)
}

fn remaining(deadline: Instant) -> io::Result<Duration> {
    deadline
        .checked_duration_since(Instant::now())
        .filter(|duration| !duration.is_zero())
        .ok_or_else(|| io::Error::new(io::ErrorKind::TimedOut, "session exchange timed out"))
}

# Architecture

## The pieces

```
┌─ Window ───────────────────────────────────────────────────┐
│  Tabs                                                       │
│  ┌─ Pane ────────────────────┬─ Pane ──────────────────────┐│
│  │  Terminal    │  Panel     │  Terminal   │  Panel       │ │
│  │  (PTY)       │            │  (PTY)      │              │ │
│  │  PANEL_ID=a3f2           │  PANEL_ID=b7c1              │ │
│  └───────────────────────────┴────────────────────────────┘ │
└─────────────────────────────────────────────────────────────┘
         ▲                ▲                    ▲
         │ PTY            │ reads panel.json   │ MCP
         │                │                    │
   buttons/keyboard  $PANEL_DIR/panel.json    MCP server
                            ▲
                            │ writes
                     the agent, flow, a CI run, a script…
```

## The three channels

### 1. State — the panel document (producer → app)

Published by whoever does the work. **The source of truth is the state the app holds per pane**,
and the transports feed it ([02-decisions.md](02-decisions.md) → D17):

| Transport | State | Who it is for |
|---|---|---|
| **File** — `$PANEL_DIR/panel.json`, overwritten whole, watched by the app | working | everybody: the barrier to entry is zero |
| **MCP** — `panel_set` hands the document over without writing the file | Phase 3 | the agent that also drives the app |
| **OSC over the PTY** | candidate, after F2.7 | whoever is behind an `ssh` or in a container with no volume |

Full specification: [04-panel-contract.md](04-panel-contract.md).

**Why the file is the default one:** it is the only one where the dumbest possible producer publishes
without writing code — `cat >` from a script — and the only one where the state survives the producer
dying without anybody programming anything. It is also read with `cat` and crosses a mounted volume
with no protocol of its own. **What it does not give:** it does not know who writes, and it does not
tell whoever published that their document was broken.

### 2. Control — MCP (agent → app)

An MCP server exposing tools to the agent running in the pane:

| Tool | What it does | Served |
|---|---|---|
| `panel_set` | Publish its pane's state | Yes, since F3's third PR |
| `panel_get` | Read its pane's last committed document | Yes |
| `panel_patch` | Change addressed parts of the document it holds | Yes |
| `panel_retire` | Intentionally end the session state this connection owns | Yes, D50 |
| `split_open` | Open a new pane | Yes, since F3's fourth PR |
| `split_read` | Read a pane's output | Yes, since F3's fourth PR |
| `split_close` | Close it | Yes, since F3's fourth PR |

All seven are advertised because all seven are answered. The three `split_*` act on **the panes this
connection opened**, named by an **opaque handle** the window hands back — sixteen random bytes,
scoped to (the pane whose channel asked, the connection) — and **the person's first byte into one of
them takes it back**: from then on the handle answers with the reason and never with the buffer.
**The revocation is an application boundary and not a confidentiality one** (F10): the process
`split_open` launched keeps reading what the person types after that, and the promise an agent author
is given is «the app stops handing you that buffer; the program you opened does not».

**The transport is a per-pane endpoint inside that pane's `0700` directory** — a Unix socket on
POSIX; on Windows, a named pipe that is **not bound until its security descriptor can be asserted**,
which is F2.7's. It carries newline-delimited JSON, and the MCP vocabulary is a binary that speaks it
rather than the channel itself.

**Security rule:** an agent reaches **its own pane**, and the consoles it opened itself, and no other.
For `panel_set` the rule is kept by a mechanism and not by a check: **the pane a call acts on is the
endpoint the connection opened**, and no message carries a pane's identifier at all. For the three
`split_*` verbs there *is* something to say — a console other than the caller's — and what is said is
an **opaque handle scoped to the connection that opened it** (D44), so reaching a console somebody
else opened is not refused, it is unsayable. See the Security table below for why a check on
`PANEL_ID` would have been worth nothing.

`panel_retire` is the explicit end of that connection's session (**D50**). It is accepted only from
the connection currently holding the panel. In one logical window transition it removes the
document —and therefore its actions and attention—, releases the hold, detaches auxiliary consoles
that connection still controls and forgets all its handles. A console the person took over stays.
The answer is sent after the panel and tree commit and before the detached pseudoterminals are
destroyed through ordinary pane and PTY destruction.

Retirement does **not** delete `panel.json`: the file cannot prove who produced it. The live pane
keeps a bounded snapshot of the file's metadata and bytes at retirement, so that unchanged file
cannot resurrect; a later valid file whose observed snapshot differs publishes normally. Closing a
pane already removes its private directory, and restoring creates a fresh identity, so the marker
does not belong in the saved layout. Connection loss is deliberately different: a crash, EOF or
accidental disconnect releases the hold but preserves the last document and every console.
Compaction is not retirement.

This repository supplies no lifecycle adapter. A harness may claim automatic clear-session support
only after a real integration run proves its explicit lifecycle event reaches `panel_retire` through
the already-connected server. Unsupported harnesses remain unsupported rather than being emulated
with terminal parsing or a fresh connection. The control endpoint, including retirement, remains
unavailable on Windows until F2.7 supplies and tests the named pipe security boundary.

### 3. Input — PTY (app → agent)

The panel's buttons write into their pane's PTY. It is the universal fallback: it works with any
program, whether it speaks MCP or not.

**Accepted risk:** a file that declares commands and a button that runs them is code execution in the
user's shell. See Security, below.

---

## Reading the output

For whatever does not speak MCP, the screen has to be read. The piece that makes that viable is
**shell marks (OSC 133)**: they delimit where the prompt starts, where the command does, and where
its output does. Without them the app sees only a river of text; with them it can bound «the last
command's output».

They are switched on by loading the shell's integration script (WezTerm already ships one; in the
app's own it will be equivalent).

**What must not be done:** parsing the painting of a text interface to deduce state. It breaks with
every version of the agent. That is what channel 1 is for.

---

## Security

| Risk | Mitigation |
|---|---|
| The panel's `actions` run commands in the shell | **The literal command is painted under its label and the click is the confirmation**, so what is on screen is what runs, character for character. `send` is plain text with no interpolation and no templates; every control character, line break, tab and zero-width scalar is **refused when the document is read, never cleaned** — cleaning would make the painted text stop being the sent text, which is the whole of this mitigation. A button that is not fully on screen is not pressable, and a press is refused on the frame the document changed |
| A link in a panel or in a console's output opens an address | **Nothing opens on its own, and nothing reaches a shell.** A panel's `line.link` opens on a click on its label; a console's plain address or OSC 8 hyperlink opens on `Ctrl`+click; hovering shows the exact address before the click — over a panel label, and over a console link while `Ctrl` is held — because an OSC 8 label may say anything and that is where its destination becomes visible. **One boundary accepts only complete `http` and `https` addresses with a host**; every other scheme, a relative path, whitespace, a control or an invisible formatting character is inert. What is kept, shown and opened is the parser's form of the address — the form the browser will load — with the characters a command interpreter reads percent-encoded, and it goes to the desktop's own opener: never to the pane's PTY, never built into a command by this application, never fetched by it. **The opener itself does build one under WSL** (`cmd.exe /c start`, escaping only `^` and `&`), which is what that encoding is for. When the opener reports it could not start anything, one line says so on standard error. And the click opens what was **painted**, not what the producer or the program says by the time the click is routed. **D51** |
| A `panel.json` of untrusted origin (cloned repo, container) | **The document's origin is unknowable and the application does not pretend otherwise.** `PANEL_DIR` is created by the application, `0700`, under the user's runtime directory and with a name drawn at random, so whoever writes there is a process of the user's own uid — the same trust level as the shell in the pane. There is nothing to test, and deducing an origin would be **D6**. What is enforceable is the premise: **the application never reads a panel document from a path it did not create** — no flag, no setting and no convenience that points it at a file somebody names. That one-line feature would be the whole of the breach |
| An agent writing into somebody else's pane | **The pane is the endpoint the connection opened, and `panel_set` carries no pane identifier for anybody to check.** The earlier wording here said the server «scopes by `PANEL_ID`», and that was not a mitigation: `PANEL_ID` is put in the child's environment, so every descendant of a pane inherits it and any process of the same uid reads it out of `/proc/<pid>/environ` — it is a name, not a credential, and it is eight hex digits drawn for collision avoidance rather than unguessability. What does the work is the directory's `0700` and the fact that there is nothing in a message to trust. **Through the file there is no scoping either**: whoever can write in the directory is the producer (D17). The three `split_*` verbs do name a console, and what they name it with is an **opaque per-connection handle** — sixteen random bytes, checked against the pair (origin pane, connection) — so a console the connection did not open cannot be said. **D44** |
| A stale console identity reaches another user's replacement socket | Before connecting, the MCP bridge, session helper and direct OpenCode plugin use `lstat` on `PANEL_DIR` and require a real directory of the effective uid with `mode & 0o077 == 0`. Final symlinks are refused, including terminal slash/dot spellings; missing paths and metadata errors fail closed. The plugin checks anew on every reconnect. Rust refusals name the directory. This leaves the check/connect pathname race and arbitrary external file producers outside the guarantee. **D7** |
| An agent reading a console the person then types into | **The handle is revoked by the person's own input, and the revocation is an application boundary rather than a confidentiality one** (F10). A keystroke or a paste, a wheel turn, a mouse report, and a held paste sent from its notice's button end it (a panel button's press does not: that is the application acting on the person's behalf). From that byte on, `split_read` and `split_close` answer with the reason, and the console's own row says it. **The program the agent opened keeps reading what the person types afterwards** — the promise is «the app stops handing you that buffer», and it is written that way wherever it is promised |
| Two producers publishing to one pane | **The first connection to publish holds the pane under a source it declares; while it does, the file is not read at all and a second source is refused by name.** The hold dies with the connection, so an agent that crashes does not freeze that panel for the life of the window. **D43** |
| A hook clearing another session's panel or consoles | **Only the connection currently holding the panel can send `panel_retire`.** A source name, `PANEL_ID`, `PANEL_DIR`, a command-hook process or an opaque console handle grants no retirement authority. The observed file snapshot is suppressed in application-owned state rather than deleted, because its producer is unknowable. **D50** |
| Reading the buffer and sending it to a model | It exfiltrates whatever is on screen: tokens, keys, API responses. Choosing what gets sent must be explicit, never automatic |
| Setting `ANTHROPIC_API_KEY` when launching a panel | Forbidden: it would change the user's billing (see D4) |
| Secrets in the panel | The panel is persisted text; it is no place for credentials |

### Declared client metadata

Native OSC7501 and the OpenCode TUI adapter update a transient pane-local signal independently of `PanelState`. The panel document, publisher hold and actions retain their existing authority. Only declared human input contributes the existing `wait` attention class; lifecycle `done` and `error` do not assert task semantics. Root native frames and one connection-owned TUI snapshot slot are bounded; child native declarations are ignored. The shared bounded OSC scanner retains OSC7 directory invalidation/reset behavior.

Measurement boundaries are unchanged: native scanning, preference application and capability-response queueing belong to the existing **interpretation** span; independent socket delivery belongs to **control** application; logo drawing/deadlines belong to **pane header** painting; stale-deadline scheduling and connection diagnostics belong to **integrated chrome**. None is attributed by subtraction to framework overhead. Hidden panes retain the same PTY pump budget. No new frame-cost improvement is claimed.

### Codex console-local startup

Ordinary Bash, Zsh and Fish consoles install a function after user startup that resolves Codex from the current PATH and calls the bundled `ivy-codex` sibling. The launcher checks client capability and isolates supported local interactive launches; MCP identity still requires the configured `env_vars` forwarding. Existing aliases and functions win. Explicit bypasses and unsupported shells remain unchanged. No global shell or Codex configuration is modified.

// OpenCode 1.18.31 TUI plugin. Only state metadata leaves this process.
import { createEffect, createRoot, createSignal } from "solid-js"
import { createConnection, type Socket } from "node:net"
import { join } from "node:path"
import { lstatSync } from "node:fs"

const envelope = (snapshot: Snapshot | null) => JSON.stringify({ tool: "agent_signal", v: 1, client: "opencode", snapshot }) + "\n"
type Snapshot = { session: string; activity: "busy" | "idle" | "unknown"; permission: boolean; question: boolean; stale?: boolean }
type Entry = { id: string; sessionID: string }

// One request in flight and one latest replacement. Read each response before
// writing the next one; neither a slow endpoint nor an outage accumulates frames.
function channel(dir: string, connected: () => void) {
  let socket: Socket | undefined, retry: ReturnType<typeof setTimeout> | undefined
  let retryReply: ReturnType<typeof setTimeout> | undefined, waitingForRefresh = false
  let pending: Snapshot | null | undefined, inflightSnapshot: Snapshot | null | undefined, inflight = false, buffer = "", stopped = false
  let reporterReady = false
  let drained: (() => void) | undefined
  function pump() {
    if (!socket || socket.connecting || inflight || pending === undefined) return
    inflight = true
    const snapshot = pending
    inflightSnapshot = snapshot
    pending = undefined
    if (snapshot === null) reporterReady = false
    socket.write(envelope(snapshot))
  }
  function connect() {
    if (stopped) return
    // A trailing slash or /. would make lstat follow a final symlink. Keep .. intact.
    let lookup = dir
    while (lookup.length > 1 && (lookup.endsWith("/") || lookup.endsWith("/."))) lookup = lookup.slice(0, -1)
    try {
      const stat = lstatSync(lookup)
      if (!stat.isDirectory() || stat.uid !== process.geteuid!() || (stat.mode & 0o077) !== 0) {
        retry = setTimeout(connect, 1000)
        return
      }
    } catch {
      retry = setTimeout(connect, 1000)
      return
    }
    const current = createConnection(join(dir, "control.sock"))
    socket = current
    reporterReady = false
    current.on("connect", () => { if (socket === current) connected() })
    current.on("data", (chunk) => {
      if (socket !== current) return
      buffer += chunk.toString()
      if (buffer.length > 65536) { current.destroy(); return }
      const end = buffer.indexOf("\n")
      if (end < 0) return
      const line = buffer.slice(0, end)
      buffer = buffer.slice(end + 1)
      try {
        const reply = JSON.parse(line)
        if (reply.ok !== true) {
          if (reply.retryable !== true) { stopped = true; current.destroy(); return }
          // The listener may be paused in the background. Reconnecting on every
          // timeout would fill its connection slots with close notifications.
          inflight = false
          inflightSnapshot = undefined
          pending = undefined
          waitingForRefresh = true
          retryReply = setTimeout(() => {
            retryReply = undefined
            if (socket !== current || stopped) return
            waitingForRefresh = false
            connected() // refresh the current route, never replay old bytes
          }, 1000)
          return
        }
      }
      catch { current.destroy(); return }
      if (inflightSnapshot && !inflightSnapshot.stale) reporterReady = true
      inflight = false
      inflightSnapshot = undefined
      pump()
      if (!inflight && pending === undefined) { drained?.(); drained = undefined }
    })
    current.on("error", () => {})
    current.on("close", () => {
      if (socket !== current) return
      if (retryReply) clearTimeout(retryReply)
      retryReply = undefined; waitingForRefresh = false
      socket = undefined; pending = undefined; inflightSnapshot = undefined; inflight = false; buffer = ""; reporterReady = false
      drained?.(); drained = undefined
      if (!stopped) retry = setTimeout(connect, 1000)
    })
  }
  connect()
  return {
    send(snapshot: Snapshot | null) {
      if (!socket || socket.connecting || (waitingForRefresh && snapshot !== null) || (snapshot?.stale && !reporterReady)) return false
      if (snapshot === null && retryReply) { clearTimeout(retryReply); retryReply = undefined; waitingForRefresh = false }
      pending = snapshot
      if (snapshot === null) reporterReady = false
      pump()
      return true
    },
    async drain() {
      if (!socket || (!inflight && pending === undefined)) return
      await new Promise<void>(resolve => {
        const timeout = setTimeout(() => { drained = undefined; resolve() }, 4000)
        drained = () => { clearTimeout(timeout); resolve() }
      })
    },
    close() { stopped = true; if (retry) clearTimeout(retry); if (retryReply) clearTimeout(retryReply); socket?.end(); socket = undefined },
  }
}

export default {
  id: "ivy.agent-signals",
  async tui(api: any) {
    const dir = process.env.PANEL_DIR
    if (!dir || !process.env.PANEL_ID || process.platform === "win32") return
    let generation = 0, stopped = false, refreshQueued = false, refreshing = false, refreshPending = false
    let timer: ReturnType<typeof setTimeout> | undefined
    let latestIds: string[] = []
    const [children, setChildren] = createSignal<string[]>([])
    let requestRefresh = () => {}
    const transport = channel(dir, () => requestRefresh())
    let lastSnapshot: Snapshot | undefined, markedStale = false
    const markStale = () => {
      if (lastSnapshot && !markedStale) {
        markedStale = transport.send({...lastSnapshot, stale: true})
      }
    }
    const withdraw = () => { lastSnapshot = undefined; markedStale = false; transport.send(null) }
    const cancelTimer = () => { if (timer) clearTimeout(timer); timer = undefined }
    // Every invalidation makes any earlier asynchronous response ineligible.
    requestRefresh = () => {
      generation++
      if (stopped) return
      const route = api.route.current
      if (route.name !== "session") withdraw()
      else if (lastSnapshot && route.params?.sessionID !== lastSnapshot.session) markStale()
      refreshPending = true
      if (refreshQueued || refreshing) return
      refreshQueued = true
      queueMicrotask(runRefresh)
    }
    async function runRefresh() {
      refreshQueued = false
      if (stopped || refreshing || !refreshPending) return
      refreshing = true
      refreshPending = false
      try { await refresh(generation) }
      finally {
        refreshing = false
        if (!stopped && refreshPending && !refreshQueued) {
          refreshQueued = true
          queueMicrotask(runRefresh)
        }
      }
    }
    async function refresh(token: number) {
      cancelTimer()
      const route = api.route.current
      if (route.name !== "session" || typeof route.params?.sessionID !== "string") {
        if (latestIds.length) { latestIds = []; setChildren([]) }; withdraw(); return
      }
      const id = route.params.sessionID
      const session = api.state.session.get(id)
      if (!api.state.ready || !session) { markStale(); return }
      const directory = session.directory
      const workspace = session.workspaceID
      try {
        const [descendants, permissions, questions] = await Promise.all([
          session.parentID ? Promise.resolve({data: []}) : api.client.session.children({sessionID: id, directory, workspace}, {throwOnError: true}),
          api.client.permission.list({directory, workspace}, {throwOnError: true}),
          api.client.question.list({directory, workspace}, {throwOnError: true}),
        ])
        if (stopped || token !== generation || api.route.current.name !== "session" || api.route.current.params?.sessionID !== id) return
        if (!Array.isArray(descendants.data) || !Array.isArray(permissions.data) || !Array.isArray(questions.data)) { markStale(); return }
        const ids = session.parentID ? [id] : [id, ...descendants.data.filter((x: any) => x.parentID === id).map((x: any) => x.id)]
        if (ids.length !== latestIds.length || ids.some((x: string, i: number) => latestIds[i] !== x)) {
          latestIds = ids
          setChildren(ids)
        }
        const visiblePermissions: Entry[] = session.parentID ? [] : ids.flatMap((x: string) => api.state.session.permission(x))
        const visibleQuestions: Entry[] = session.parentID ? [] : ids.flatMap((x: string) => api.state.session.question(x))
        const pendingPermissions: Entry[] = permissions.data.filter((x: Entry) => ids.includes(x.sessionID))
        const pendingQuestions: Entry[] = questions.data.filter((x: Entry) => ids.includes(x.sessionID))
        // A missing getter entry has not been hydrated; never turn it into a clear.
        const missing = (pending: Entry[], visible: Entry[]) => pending.some(p => !visible.some(v => v.id === p.id && v.sessionID === p.sessionID))
        if (!session.parentID && (missing(pendingPermissions, visiblePermissions) || missing(pendingQuestions, visibleQuestions))) {
          markStale(); timer = setTimeout(requestRefresh, 2000); return
        }
        const exists = (visible: Entry[], pending: Entry[]) => visible.some(v => pending.some(p => p.id === v.id && p.sessionID === v.sessionID))
        const permission = exists(visiblePermissions, pendingPermissions)
        const question = exists(visibleQuestions, pendingQuestions)
        const status = api.state.session.status(id)?.type
        const activity = status === "busy" || status === "retry" ? "busy" : status === "idle" ? "idle" : "unknown"
        lastSnapshot = {session: id, activity, permission, question}
        markedStale = false
        transport.send(lastSnapshot)
        if (permission || question) timer = setTimeout(requestRefresh, 2000)
      } catch {
        // Keep the last claim; a failed refresh is not an answer.
        if (!stopped && token === generation) { markStale(); timer = setTimeout(requestRefresh, 2000) }
      }
    }
    const dispose = createRoot((dispose) => {
      createEffect(() => {
        const route = api.route.current
        const id = route.name === "session" ? route.params?.sessionID : undefined
        api.state.ready
        if (typeof id === "string") {
          api.state.session.get(id); api.state.session.status(id)
          for (const child of children()) { api.state.session.permission(child).length; api.state.session.question(child).length }
        }
        requestRefresh()
      })
      return dispose
    })
    const unsubscribe = ["session.created", "session.updated", "session.deleted", "session.status", "permission.asked", "permission.replied", "question.asked", "question.replied", "question.rejected"].map(
      type => api.event.on(type, () => requestRefresh()),
    )
    api.lifecycle.onDispose(async () => {
      stopped = true; generation++; cancelTimer(); dispose(); unsubscribe.forEach((off: () => void) => off())
      withdraw()
      // Drain ordered frames within OpenCode’s five-second cleanup budget.
      await transport.drain()
      transport.close()
    })
  },
}
